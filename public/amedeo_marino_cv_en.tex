%--------CUSTOM COMMANDS--------------
% resumeItem renders a bullet point
\newcommand{\resumeItem}[1]{
  \item\bodyfs #1
}

\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=1.4em, rightmargin=0pt, itemsep=1.5pt plus 1pt, topsep=2.5pt plus 1pt, parsep=0pt, partopsep=0pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}}

% resumeSectionType: bold label in a fixed-width box so the ":" line up
\newcommand{\resumeSectionType}[3]{
  \item\bodyfs\makebox[0.15\linewidth][l]{\textbf{#1}}#2\hspace{0.7em}#3
}

% resumeTrioHeading: three elements, second italicised, first bold
\newcommand{\resumeTrioHeading}[3]{
  \item\bodyfs
    \begin{tabular*}{\linewidth}[t]{
      l@{\extracolsep{\fill}}c@{\extracolsep{\fill}}r
    }
      \textbf{#1} & \textit{#2} & #3
    \end{tabular*}
}

% resumeQuadHeading: two rows, first bold, second italic
\newcommand{\resumeQuadHeading}[4]{
  \item\headfs
  \begin{tabular*}{\linewidth}[t]{l@{\extracolsep{\fill}}r}
    \textbf{#1} & {\bodyfs #2} \\
    \textit{\subfs #3} & \textit{\subfs #4} \\
  \end{tabular*}
}

\newcommand{\resumeQuadHeadingChild}[2]{
  \item
  \begin{tabular*}{\linewidth}[t]{l@{\extracolsep{\fill}}r}
    \textbf{\small#1} & {\small#2} \\
  \end{tabular*}
}

% itemsep carries the stretch that distributes spare vertical space
\newcommand{\resumeHeadingListStart}{
  \begin{itemize}[leftmargin=0pt, label={}, itemsep=6pt plus 3pt minus 1pt, topsep=0pt, parsep=0pt, partopsep=0pt]
}
\newcommand{\resumeHeadingListEnd}{\end{itemize}}
% tighter list for the skills rows
\newcommand{\resumeSkillListStart}{
  \begin{itemize}[leftmargin=0pt, label={}, itemsep=3pt plus 1pt, topsep=0pt, parsep=0pt, partopsep=0pt]
}
